interface ITeclaMenu {
  key: string;
  shift: boolean;
  atual: number;
  max: number;
}
 
export function larguraPorTecla({ key, shift, atual, max }: ITeclaMenu): number | null {
  const passo = shift ? 48 : 16;
  switch (key) {
    case 'ArrowRight':
      return atual === MENU_TRILHO ? MENU_MIN : Math.min(max, atual + passo);
    case 'ArrowLeft':
      return atual - passo < MENU_MIN ? 0 : atual - passo;
    case 'Home':
      return 0;
    case 'End':
      return max;
    case 'Enter':
      return atual === MENU_TRILHO ? MENU_PADRAO : MENU_TRILHO;
    default:
      return null;
  }
}
